// ============================================================================
// Configuration Types (YAML Parsing)
// Location: internal/config/config.go

// WorkflowConfig represents the workflow section in config.yaml
type WorkflowConfig struct {
	Steps  []WorkflowStepConfig `yaml:"steps"`
	Review ReviewStepConfig     `yaml:"review"`
}

// WorkflowStepConfig represents a single custom workflow step definition in config.yaml
type WorkflowStepConfig struct {
	Name    string `yaml:"name"`
	Command string `yaml:"command"`
	Retries int    `yaml:"retries,omitempty"` // Default: 1
}

// ReviewStepConfig represents the mandatory review step configuration in config.yaml
type ReviewStepConfig struct {
	Command string `yaml:"command"`
	Retries int    `yaml:"retries,omitempty"` // Default: 1
}

// ProjectConfig extension (add Workflow field to existing struct)
type ProjectConfig struct {
	Agent            *string         `yaml:"agent"`
	AgentArgs        []string        `yaml:"agent_args"`
	ImplementRetries *int            `yaml:"implement_retries"`
	CommitRetries    *int            `yaml:"commit_retries"`
	Iterations       *int            `yaml:"iterations"`
	Commands         *Commands       `yaml:"commands"`
	Workflow         *WorkflowConfig `yaml:"workflow"` // NEW FIELD
}

// HomeConfig extension (add Workflow field to existing struct)
type HomeConfig struct {
	Agent            *string         `yaml:"agent"`
	AgentArgs        []string        `yaml:"agent_args"`
	ImplementRetries *int            `yaml:"implement_retries"`
	CommitRetries    *int            `yaml:"commit_retries"`
	Iterations       *int            `yaml:"iterations"`
	Commands         *Commands       `yaml:"commands"`
	Workflow         *WorkflowConfig `yaml:"workflow"` // NEW FIELD
}

// ============================================================================
// Runtime Types (Workflow Execution)
// Location: internal/types/types.go
// ============================================================================

// WorkflowStep represents a workflow step at runtime (after config parsing and validation)
type WorkflowStep struct {
	Name            string // Step name (e.g., "implement", "design", "test", "review")
	CommandFilePath string // Resolved absolute file path to command file
	Retries         int    // Maximum retry attempts (0 = infinite)
	IsReview        bool   // True if this is the review step (exit gate)
	Order           int    // Execution order (0-based index)
}

// Workflow represents the complete workflow at runtime
type Workflow struct {
	Steps            []WorkflowStep // Ordered list of all workflow steps (custom + review)
	MaxIterations    int            // Maximum development iterations (0 = infinite)
	CurrentIteration int            // Current iteration number (1-based)
}

// Config extension (add Workflow field to existing struct)
type Config struct {
	Agent         string
	AgentArgs     []string
	TaskFilePath  string
	SessionID     string
	SessionRoot   string
	CommandFiles  *CommandFiles
	Workflow      *Workflow // NEW FIELD
	OutputFormat  string    // "text" or "json" (for FR-021)

	// Deprecated fields (kept for migration, will be removed after workflow refactor)
	MaxImplementRetries int // DEPRECATED: Use Workflow.Steps[].Retries instead
	MaxCommitRetries    int // DEPRECATED: Use Workflow.Steps[].Retries instead
	MaxReviewCycles     int // DEPRECATED: Use Workflow.MaxIterations instead
}
